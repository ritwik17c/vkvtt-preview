rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function authEmail() {
      return signedIn() ? request.auth.token.get('email', '') : '';
    }

    function verifiedEmail() {
      return signedIn()
             && request.auth.token.get('email_verified', false) == true
             && authEmail() is string
             && authEmail().size() > 0;
    }

    function hasUidProfile() {
      return signedIn()
             && exists(/databases/$(database)/documents/authorizedUsers/$(request.auth.uid));
    }

    function uidProfile() {
      return get(/databases/$(database)/documents/authorizedUsers/$(request.auth.uid));
    }

    function uidActive() {
      return hasUidProfile() && uidProfile().data.active == true;
    }

    function uidRole() {
      return uidActive() ? uidProfile().data.role : '';
    }

    function viewerAllowed() {
      return verifiedEmail()
             && exists(/databases/$(database)/documents/viewerEmails/$(authEmail()))
             && get(/databases/$(database)/documents/viewerEmails/$(authEmail())).data.active == true;
    }

    function canReadSchool() {
      return uidActive() || viewerAllowed();
    }

    function isAdmin() {
      return uidActive() && uidRole() == 'admin';
    }

    function isLeaveEditor() {
      return uidActive() && uidRole() == 'leave_editor';
    }

    function canPrepareLeave() {
      return isAdmin() || isLeaveEditor();
    }

    // Timetable Studio is available to the Principal/Admin and to individual
    // UID accounts explicitly delegated through User Access & Roles.
    function canUseTimetableStudio() {
      return uidActive()
             && (uidRole() == 'admin'
                 || uidProfile().data.get('permissions', {}).get('timetableStudio', false) == true);
    }

    function isStatusEditor() {
      return uidActive() && uidRole() in ['admin', 'manager'];
    }

    function isProxyManager() {
      return uidActive() && uidRole() == 'proxy_manager';
    }

    function isLeaveViewer() {
      return (uidActive() && uidRole() == 'leave_viewer')
             || (verifiedEmail()
                 && exists(/databases/$(database)/documents/viewerEmails/$(authEmail()))
                 && get(/databases/$(database)/documents/viewerEmails/$(authEmail())).data.active == true
                 && get(/databases/$(database)/documents/viewerEmails/$(authEmail())).data.role == 'leave_viewer');
    }

    function leaveHistoryVisible() {
      return exists(/databases/$(database)/documents/leaveControl/current)
             && get(/databases/$(database)/documents/leaveControl/current).data.myAreaLeaveHistoryVisible == true;
    }

    function canPublishProxy() {
      return isStatusEditor() || isProxyManager();
    }

    function isAttendanceManager() {
      return uidActive() && uidRole() in ['admin', 'manager', 'attendance_manager'];
    }

    function attendanceOwner() {
      return signedIn() && request.resource.data.uid == request.auth.uid;
    }

    function proxyOnlyUpdate() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
        'allotments',
        'supervisions',
        'updatedAt',
        'updatedBy',
        'updatedByEmail'
      ])
      && request.resource.data.updatedBy == request.auth.uid;
    }

    function proxyOnlyCreate(date) {
      return request.resource.data.keys().hasOnly([
        'date',
        'dayName',
        'allotments',
        'supervisions',
        'updatedAt',
        'updatedBy',
        'updatedByEmail'
      ])
      && request.resource.data.keys().hasAll([
        'date',
        'dayName',
        'allotments',
        'supervisions',
        'updatedBy'
      ])
      && request.resource.data.date == date
      && request.resource.data.updatedBy == request.auth.uid;
    }

    match /authorizedUsers/{uid} {
      allow get: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow list, create, update, delete: if isAdmin();
    }

    match /viewerEmails/{email} {
      allow get: if isAdmin() || (verifiedEmail() && authEmail() == email);
      allow list, create, update, delete: if isAdmin();
    }

    match /master/{documentId} {
      allow read: if canReadSchool();
      allow create, update, delete: if isAdmin();
    }

    match /annualCalendar/{documentId} {
      allow read: if isAdmin()
                  || (canReadSchool()
                      && resource.data.get('visible', true) == true);
      allow create, update, delete: if isAdmin();
    }

    match /timetableVersions/{versionId} {
      allow read: if canUseTimetableStudio();
      allow create: if canUseTimetableStudio()
                    && request.resource.data.status in ['draft', 'ready', 'inactive'];
      allow update: if isAdmin()
                    || (canUseTimetableStudio()
                        && resource.data.status != 'active'
                        && request.resource.data.status in ['draft', 'ready', 'inactive']);
      allow delete: if isAdmin() && resource.data.status != 'active';
    }

    match /timetableActivations/{activationId} {
      allow read: if canUseTimetableStudio();
      allow create: if isAdmin()
                    && request.resource.data.activatedByUid == request.auth.uid;
      allow update, delete: if false;
    }

    match /dailyRecords/{date} {
      allow read: if uidActive() && uidRole() in ['admin', 'manager', 'proxy_manager'];
      allow create: if isStatusEditor()
                    || (isProxyManager() && proxyOnlyCreate(date));
      allow update: if isStatusEditor()
                    || (isProxyManager() && proxyOnlyUpdate());
      allow delete: if isAdmin();
    }

    match /publishedProxy/{date} {
      allow read: if canReadSchool();
      allow create, update: if isAdmin()
                            || (canPublishProxy()
                                && request.resource.data.date == date
                                && request.resource.data.finalizedByUid == request.auth.uid);
      allow delete: if isAdmin();
    }

    match /proxyEditAuthorizations/{date} {
      allow read: if uidActive() && uidRole() in ['admin', 'manager', 'proxy_manager'];
      allow create, update: if isAdmin()
                            && request.resource.data.date == date
                            && request.resource.data.enabled == true
                            && request.resource.data.enabledByUid == request.auth.uid;
      allow delete: if isAdmin();
    }

    // Approved leave projections stay approval-controlled. Leave Editors get
    // read-only context for balance/history; writes remain Status Editor only.
    match /approvedStatusPlans/{planId} {
      allow read: if isAdmin() || isLeaveViewer() || isLeaveEditor();
      allow create, update, delete: if isStatusEditor();
    }

    match /approvedDailyStatus/{date} {
      allow read: if isAdmin() || isLeaveViewer();
      allow create, update, delete: if isStatusEditor();
    }

    // Phase-2 delegated preparation. Leave Editors can prepare/correct only
    // their own provisional/returned work. Principal/Admin controls approval.
    match /provisionalLeavePlans/{planId} {
      allow read: if isAdmin()
                  || isLeaveEditor()
                  || isProxyManager()
                  || isStatusEditor();

      allow create: if canPrepareLeave()
                    && request.resource.data.approvalStatus == 'provisional'
                    && request.resource.data.createdByUid == request.auth.uid
                    && request.resource.data.updatedByUid == request.auth.uid
                    && !request.resource.data.keys().hasAny([
                      'approvedByUid','approvedByEmail','approvedAt','approvedAtMs',
                      'returnedByUid','returnedByEmail','returnedAt','returnedAtMs',
                      'returnRemark','cancelledByUid','cancelledByEmail',
                      'cancelledAt','cancelledAtMs'
                    ]);

      allow update: if isAdmin()
                    || (isLeaveEditor()
                        && resource.data.createdByUid == request.auth.uid
                        && resource.data.approvalStatus in ['provisional', 'returned']
                        && request.resource.data.approvalStatus == 'provisional'
                        && request.resource.data.createdByUid == resource.data.createdByUid
                        && request.resource.data.createdByEmail == resource.data.createdByEmail
                        && request.resource.data.createdAtMs == resource.data.createdAtMs
                        && request.resource.data.updatedByUid == request.auth.uid
                        && !request.resource.data.diff(resource.data).affectedKeys().hasAny([
                          'approvedByUid','approvedByEmail','approvedAt','approvedAtMs',
                          'returnedByUid','returnedByEmail','returnedAt','returnedAtMs',
                          'cancelledByUid','cancelledByEmail','cancelledAt','cancelledAtMs'
                        ]));

      allow delete: if false;
    }

    // Immutable Principal/Admin decision trail.
    match /leaveApprovalAudit/{auditId} {
      allow read: if isAdmin();
      allow create: if isAdmin()
                    && request.resource.data.action in ['approved','returned','cancelled']
                    && request.resource.data.actedByUid == request.auth.uid;
      allow update, delete: if false;
    }

    match /personalStatus/{email}/records/{date} {
      allow read: if (verifiedEmail() && authEmail() == email && canReadSchool() && leaveHistoryVisible())
                  || isStatusEditor();
      allow create, update, delete: if isStatusEditor();
    }

    match /leaveControl/{documentId} {
      allow read: if canReadSchool();
      allow create, update, delete: if isAdmin();
    }

    match /leaveAudit/{auditId} {
      allow read, create, update, delete: if isAdmin();
    }

    match /legacyLeaveAccounting/{entryId} {
      allow read: if isAdmin();
      allow create, update, delete: if isAdmin();
    }

    // Leave Editors may read policy required to validate provisional entries;
    // only Principal/Admin can alter policy.
    match /leaveRules/{documentId} {
      allow read: if isStatusEditor() || isLeaveEditor();
      allow create, update: if isAdmin();
      allow delete: if false;
    }

    match /attendanceSettings/{documentId} {
      allow read: if canReadSchool();
      allow create, update, delete: if isAdmin();
    }

    match /attendanceSchedules/{scheduleId} {
      allow read: if canReadSchool();
      allow create, update, delete: if isAdmin();
    }

    match /attendanceEvents/{eventId} {
      allow get: if isAttendanceManager()
                 || (canReadSchool() && resource.data.uid == request.auth.uid);
      allow list: if isAttendanceManager()
                  || (canReadSchool() && resource.data.uid == request.auth.uid);
      allow create: if canReadSchool()
                    && attendanceOwner()
                    && request.resource.data.keys().hasAll([
                      'uid', 'date', 'type', 'recordedAt', 'latitude', 'longitude',
                      'accuracy', 'vpnMockAcknowledged', 'verificationStatus',
                      'scheduleSnapshot'
                    ])
                    && request.resource.data.type in [
                      'arrival', 'leave', 'return', 'final',
                      'od_check_in', 'od_check_out'
                    ]
                    && request.resource.data.verificationStatus == 'provisional'
                    && request.resource.data.vpnMockAcknowledged == true;
      allow update, delete: if isAdmin();
    }

    match /attendanceTestAudit/{auditId} {
      allow read, create: if isAdmin();
      allow update, delete: if false;
    }

    match /attendanceLate/{lateId} {
      allow read: if canReadSchool();
      allow create: if canReadSchool()
                    && attendanceOwner()
                    && request.resource.data.keys().hasOnly([
                      'date', 'uid', 'staffName', 'staffGroup', 'arrivalTime',
                      'arrivalClientTimeMs', 'lateMinutes',
                      'lateThresholdMinutes', 'verificationStatus'
                    ])
                    && request.resource.data.verificationStatus == 'provisional';
      allow update, delete: if isAttendanceManager();
    }

    match /odAssignments/{assignmentId} {
      allow get, list: if isAttendanceManager()
                       || (canReadSchool()
                           && (resource.data.staffUid == request.auth.uid
                               || (verifiedEmail() && resource.data.staffEmail == authEmail())));
      allow create, update, delete: if isAttendanceManager();
    }

    match /biometricImports/{batchId} {
      allow get, list: if isAttendanceManager();
      allow create: if isAttendanceManager()
                    && request.resource.data.importedBy == request.auth.uid
                    && request.resource.data.status == 'active';
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    match /biometricRecords/{recordId} {
      allow get, list: if isAttendanceManager();
      allow create: if isAttendanceManager()
                    && request.resource.data.importedBy == request.auth.uid
                    && request.resource.data.batchId is string;
      allow update: if false;
      allow delete: if isAdmin();
    }

    match /attendanceVerifications/{eventId} {
      allow get, list: if isAttendanceManager()
                       || (canReadSchool() && resource.data.uid == request.auth.uid);
      allow create, update: if isAttendanceManager()
                            && request.resource.data.eventId == eventId
                            && request.resource.data.verifiedBy == request.auth.uid
                            && request.resource.data.status in [
                              'verified', 'verified_correction', 'app_only',
                              'biometric_only', 'mismatch'
                            ];
      allow delete: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
